cmake_minimum_required(VERSION 3.16)
project(FractalModels LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BUILD_APP "Compilar el controlador con OpenCV" ON)

add_library(fractal_codec STATIC src/codec.cpp src/color.cpp src/execution.cpp fine-grained/encoder.cpp coarse-grained/encoder.cpp src/cooperative_encoder.cpp src/cpu_topology.cpp CMP/encoder.cpp SMT/encoder.cpp src/parallel_encoder.cpp)
find_package(Threads REQUIRED)
target_link_libraries(fractal_codec PUBLIC Threads::Threads)
target_include_directories(fractal_codec PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
enable_testing()
if(BUILD_APP)
    find_package(OpenCV 4 REQUIRED COMPONENTS core imgcodecs)
    add_library(fractal_app STATIC app/cli.cpp app/file_io.cpp app/color_io.cpp app/image_io.cpp)
    target_include_directories(fractal_app PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/app" PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(fractal_app PUBLIC fractal_codec PRIVATE ${OpenCV_LIBS})
endif()
add_subdirectory(sequential)
add_subdirectory(fine-grained)
add_subdirectory(coarse-grained)
add_subdirectory(CMP)
add_subdirectory(SMT)

add_executable(codec_tests tests/test_codec.cpp app/file_io.cpp)
target_include_directories(codec_tests PRIVATE app)
target_link_libraries(codec_tests PRIVATE fractal_codec)
add_test(NAME codec_tests COMMAND codec_tests)
add_executable(color_tests tests/test_color.cpp app/color_io.cpp app/file_io.cpp)
target_include_directories(color_tests PRIVATE app)
target_link_libraries(color_tests PRIVATE fractal_codec)
add_test(NAME color_tests COMMAND color_tests)
add_executable(execution_tests tests/test_execution.cpp)
target_link_libraries(execution_tests PRIVATE fractal_codec)
add_test(NAME execution_tests COMMAND execution_tests)
add_executable(fine_grained_tests tests/test_fine_grained.cpp)
target_link_libraries(fine_grained_tests PRIVATE fractal_codec)
add_test(NAME fine_grained_tests COMMAND fine_grained_tests)
set_tests_properties(fine_grained_tests PROPERTIES TIMEOUT 45)
add_executable(coarse_grained_tests tests/test_coarse_grained.cpp)
target_link_libraries(coarse_grained_tests PRIVATE fractal_codec)
add_test(NAME coarse_grained_tests COMMAND coarse_grained_tests)
set_tests_properties(coarse_grained_tests PROPERTIES TIMEOUT 45)
add_executable(cmp_tests tests/test_cmp.cpp)
target_link_libraries(cmp_tests PRIVATE fractal_codec)
add_test(NAME cmp_tests COMMAND cmp_tests)
set_tests_properties(cmp_tests PROPERTIES TIMEOUT 45)
add_test(NAME cmp_hardware_tests COMMAND cmp_tests --hardware)
set_tests_properties(cmp_hardware_tests PROPERTIES TIMEOUT 45 SKIP_RETURN_CODE 77)
add_executable(smt_tests tests/test_smt.cpp)
target_link_libraries(smt_tests PRIVATE fractal_codec)
add_test(NAME smt_tests COMMAND smt_tests)
set_tests_properties(smt_tests PROPERTIES TIMEOUT 45)
add_test(NAME smt_hardware_tests COMMAND smt_tests --hardware)
set_tests_properties(smt_hardware_tests PROPERTIES TIMEOUT 90 SKIP_RETURN_CODE 77)
if(BUILD_APP)
    add_executable(image_io_tests tests/test_image_io.cpp app/image_io.cpp)

    target_link_libraries(image_io_tests PRIVATE fractal_codec ${OpenCV_LIBS})

        endif()

        set_tests_properties(benchmark_report_tests PROPERTIES TIMEOUT 60)
        add_test(NAME benchmark_pipeline_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_benchmark_pipeline.py" --build-dir "${CMAKE_BINARY_DIR}")
        set_tests_properties(benchmark_pipeline_tests PROPERTIES TIMEOUT 120)
        add_test(NAME perf_parser_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_perf.py" --build-dir "${CMAKE_BINARY_DIR}")
        add_test(NAME workflow_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_workflow.py")
        set_tests_properties(workflow_tests PROPERTIES TIMEOUT 30)
        add_test(NAME workflow_report_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_workflow.py" --integration)
        set_tests_properties(workflow_report_tests PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77)

        add_test(NAME model_runner_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_model_runner.py")
        set_tests_properties(model_runner_tests PROPERTIES TIMEOUT 20)
        add_test(NAME models_integration_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/test_models.py" --build-dir "${CMAKE_BINARY_DIR}" --results-dir "${CMAKE_BINARY_DIR}/results/integration" --models sequential fine-grained coarse-grained --quick)
        set_tests_properties(models_integration_tests PROPERTIES TIMEOUT 120)
        add_test(NAME smt_record_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_smt_records.py")
        add_test(NAME cli_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py" "$<TARGET_FILE:fractal_seq>" "$<TARGET_FILE:fractal_fine>" "$<TARGET_FILE:fractal_coarse>" "$<TARGET_FILE:fractal_cmp>" "$<TARGET_FILE:fractal_smt>")
    endif()
endif()
