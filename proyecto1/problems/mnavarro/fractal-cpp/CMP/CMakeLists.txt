cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND NOT TARGET fractal_codec)
    project(FractalCMP LANGUAGES CXX)
    enable_testing()
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." "${CMAKE_CURRENT_BINARY_DIR}/shared")
else()
    if(BUILD_APP)
        add_executable(fractal_cmp main.cpp)
        target_link_libraries(fractal_cmp PRIVATE fractal_app)
        set_target_properties(fractal_cmp PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    endif()
    add_subdirectory(local)
endif()
