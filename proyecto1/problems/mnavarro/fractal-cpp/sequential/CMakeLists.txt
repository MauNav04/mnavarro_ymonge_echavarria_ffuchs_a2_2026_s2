cmake_minimum_required(VERSION 3.16)
# Mantiene cmake -S sequential: la única biblioteca compartida se construye desde la raíz.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND NOT TARGET fractal_codec)
    project(FractalSequential LANGUAGES CXX)
    enable_testing()
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." "${CMAKE_CURRENT_BINARY_DIR}/shared")
else()
    if(BUILD_APP)
        add_executable(fractal main.cpp ../app/file_io.cpp ../app/color_io.cpp ../app/image_io.cpp)
        target_include_directories(fractal PRIVATE ../app ${OpenCV_INCLUDE_DIRS})
        target_link_libraries(fractal PRIVATE fractal_codec ${OpenCV_LIBS})
        set_target_properties(fractal PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    endif()
endif()
